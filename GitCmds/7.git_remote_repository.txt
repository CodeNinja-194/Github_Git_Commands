//  Working with Remote Repositories - Push, Pull, and Synchronization

//  INITIAL SETUP: CREATING SSH KEY (Optional but Recommended)

1:  ssh-keygen -t rsa -b 4096 -C "your-email@example.com"
    //  Generates SSH key pair (public and private)
    //  RSA algorithm with 4096 bits (strong cryptography)

2:  eval $(ssh-agent -s)
    //  Starts the SSH agent (authentication manager)

3:  ssh-add ~/.ssh/id_rsa
    //  Adds your private key to SSH agent

4:  cat ~/.ssh/id_rsa.pub
    //  Displays your public SSH key
    //  Copy this and paste into GitHub Settings > SSH Keys

//  FOR MAC USERS:
5:  pbcopy < ~/.ssh/id_rsa.pub
    //  Copies public key to clipboard directly

//  FOR WINDOWS USERS:
6:  clip < ~/.ssh/id_rsa.pub
    //  Copies public key to clipboard (Windows command)

//  BASIC REMOTE OPERATIONS

7:  git remote add origin <repository-url>
    //  Connects your local repository to remote (typically GitHub)
    //  'origin' is the standard name for the main remote

8:  git remote
    //  Lists all configured remotes (shows remote names)

9:  git remote -v
    //  Shows all remotes with their URLs (fetch and push)
    //  Output example:
    //  origin  https://github.com/user/repo.git (fetch)
    //  origin  https://github.com/user/repo.git (push)

10: git remote rename origin upstream
    //  Renames a remote (useful for reorganizing)

11: git remote remove origin
    //  Removes a remote connection

12: git remote show origin
    //  Shows detailed information about the remote (URL, branches, etc.)

//  HTTPS vs SSH

13: git remote add origin https://github.com/user/repo.git
    //  Uses HTTPS (requires password or personal access token)
    //  Easier to set up but slower

14: git remote add origin git@github.com:user/repo.git
    //  Uses SSH (requires SSH key setup)
    //  Faster and more secure, but requires key configuration

//  CHANGING REMOTE URL

15: git remote set-url origin <new-url>
    //  Changes the URL of existing remote

16: git remote set-url origin https://github.com/user/new-repo.git
    //  Example: switching to different repository

17: git remote set-url origin git@github.com:user/repo.git
    //  Example: switching from HTTPS to SSH

//  PUSHING COMMITS

18: git push
    //  Pushes to default remote (usually origin) and current branch

19: git push origin main
    //  Pushes current branch to origin/main on remote

20: git push -u origin main
    //  Pushes and sets upstream branch (first push to new branch)
    //  -u flag sets tracking relationship

21: git push origin branch_name
    //  Pushes specific branch to remote

22: git push origin feature:main
    //  Pushes 'feature' branch but names it 'main' on remote

23: git push --all
    //  Pushes all local branches to remote

24: git push --tags
    //  Pushes all tags to remote

25: git push --force
    //  Force pushes (CAREFUL! Only on personal branches, never on shared!)

26: git push -f origin branch_name
    //  Force push specific branch

//  PULLING CHANGES

27: git pull
    //  Fetches and merges changes from default remote

28: git pull origin main
    //  Fetches and merges specific branch from remote

29: git pull --rebase
    //  Fetches and rebases instead of merging (cleaner history)

//  FETCHING CHANGES (Download without merging)

30: git fetch
    //  Downloads changes from remote without modifying working directory

31: git fetch origin
    //  Fetches from specific remote

32: git fetch --all
    //  Fetches from all configured remotes

33: git fetch origin branch_name
    //  Fetches specific branch only

//  HANDLING COMMON ERRORS

34: //  ERROR: rejected - need to fetch first
    git pull origin main
    //  Solution: Pull latest changes before pushing

35: //  ERROR: fatal - no upstream branch
    git push -u origin branch_name
    //  Solution: Use -u flag to set upstream on first push

36: //  ERROR: could not resolve host
    //  Check internet connection or SSH configuration

//  DELETING REMOTE BRANCHES

37: git push -d origin branch_name
    //  Deletes branch from remote repository

38: git push origin --delete branch_name
    //  Alternative syntax for deleting remote branch

39: git branch -dr origin/branch_name
    //  Removes local tracking branch for deleted remote branch

//  TRACKING REMOTE BRANCHES

40: git branch -r
    //  Lists remote tracking branches (like origin/main, origin/develop)

41: git checkout --track origin/branch_name
    //  Creates local branch tracking remote branch

42: git branch -u origin/main
    //  Sets upstream branch for current local branch

//  WORKFLOW EXAMPLE

//  1. Clone repository:
//     git clone https://github.com/user/repo.git

//  2. Create feature branch:
//     git checkout -b feature/new-feature

//  3. Make changes and commit:
//     git add .
//     git commit -m "Add new feature"

//  4. Push to remote:
//     git push -u origin feature/new-feature

//  5. On GitHub, create Pull Request

//  6. After approval, merge and delete remote branch:
//     git push -d origin feature/new-feature

//  NOTES:
//  - SSH is recommended over HTTPS for speed and security
//  - Always pull before pushing to avoid conflicts
//  - Use -u flag on first push to set upstream
//  - Never force push to main/master branches
//  - origin is the default remote name
//  - Always verify remote URL with 'git remote -v'
